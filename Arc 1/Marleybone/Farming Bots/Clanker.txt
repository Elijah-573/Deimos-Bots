block handle_load {
    until not loading {
        sleep 4
    }
}

block enter {
    sleep 2
    tp XYZ(-2105.936767578125, 1783.5013427734375, -239.99899291992188)
    sleep 0.3
    walkto XYZ(-2186.6259765625, 2228.6328125, -220.08297729492188)
    sleep 2

        sendkey X, 0.1
    }
    sleep 0.1
    waitforzonechange completion
}

block clanker {
    #Clanker
    tp XYZ(-22773.078125, 19476.826171875, 3.0517578125e-05)
    sleep 0.1
    walkto XYZ(-22759.46875, 20187.73828125, 36.30755615234375)

}


block quit {
    sleep 2
    mass sendkey ESC 
    sleep 2
    if windowvisible ['WorldView', 'DeckConfiguration', 'SettingPage', 'QuitButton'] {
        clickwindow ['WorldView', 'DeckConfiguration', 'SettingPage', 'QuitButton']
    }
    sleep 2
    if windowvisible ['MessageBoxModalWindow', 'messageBoxBG', 'messageBoxLayout', 'AdjustmentWindow', 'Layout', 'centerButton'] {
        clickwindow ['MessageBoxModalWindow', 'messageBoxBG', 'messageBoxLayout', 'AdjustmentWindow', 'Layout', 'centerButton']
    }
    call handle_load
}

block relog {
    sleep 2
    waitforpath ['WorldView', 'mainWindow']
    sleep 2
    clickwindow ['WorldView', 'mainWindow', 'btnPlay']
    call handle_load
}



call enter 
call fireguts
call puzzle 
call clanker 
call quit 
call relog

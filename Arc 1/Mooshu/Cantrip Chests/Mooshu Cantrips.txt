block P1_Cast_Magic_Touch_At_Cantrip {

block P1_Check_If_Cantrip_Chest_Is_Active {
	times 3 {
		print "Checking if cantrip chest is active..."

			p1 entitytp RitualObject_MS
			call P1_Cast_Magic_Touch_At_Cantrip
			break

		}

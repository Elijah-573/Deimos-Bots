	while p1 loading {
		sleep 2
	}
	sleep 1
	if NOT p1 loading {
		if NOT p1 incombat {

				sleep .1
			}

block Quest_Follow_Checker_P1 {
	if NOT p1 windowvisible ['WorldView', 'windowHUD', 'QuestHelperHud', 'ElementWindow'] {
		while NOT p1 windowvisible ['WorldView', 'DeckConfiguration', 'wndQuestList', 'QuestLogMainlineAndSideButton'] {
			p1 sendkey Q, .5
			sleep 1
		}
		times 2 {
			p1 clickwindow ['WorldView', 'DeckConfiguration', 'wndQuestList', 'QuestLogMainlineAndSideButton']
		}
		if p1 windowvisible ['WorldView', 'DeckConfiguration', 'wndQuestList', 'wndQuestInfo0', 'questInfoWindow', 'wndQuestInfo', 'LeftMainline'] {
			times 2 {
				p1 clickwindow ['WorldView', 'DeckConfiguration', 'wndQuestList', 'wndQuestInfo0']
			}
		} else {
			print "Unable to select main quest. Please select main quest."
		}
		while p1 windowvisible ['WorldView', 'DeckConfiguration', 'wndQuestList', 'QuestLogMainlineAndSideButton'] {
			p1 sendkey Q, .5
			sleep 1
		}
	}
}

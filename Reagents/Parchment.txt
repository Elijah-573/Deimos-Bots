###deimos_expertmode 
block load {
    # 5 seconds for loading to pass, and handles loading if occurs
    times 5 {
        if p1 loading {
            break
        } else {
            sleep 1
        }
    }
    while p1 loading {
        sleep 1.25
    }
}

block dialogue {
    block dialoguestall {
        times 4 {
            if p1 windowvisible ['WorldView', 'wndDialogMain'] {
                break
            } else {
                sleep 0.25
            }
        }
    }
    call dialoguestall
    block dialoguebusiness {
        while p1 windowvisible ['WorldView', 'wndDialogMain'] {
            sendkey SPACEBAR, 0.1
            sleep 0.1
        }
    }
    call dialoguebusiness
}

block xwork {
    # 2 seconds for the reagent prompt to pass, and handles business if it occcurs
    times 4 {
        if p1 windowvisible ['WorldView', 'NPCRangeWin', 'imgBackground'] {
            break
        } else {
            sleep 0.5
        }
    }
    while p1 windowvisible ['WorldView', 'NPCRangeWin', 'imgBackground'] {
        p1 sendkey X, 0.1 
    }
}

block reagent {
    loop {
        if p1 inrange "Parchment" {
            print parchment detected 
            p1 entitytp Parchment
            call dialogue 
            call xwork
            sleep 0.5
        } else {
            break
        }
    }
}

if p1 inzone Krokotopia/KT_Pyramid/KT_PalaceOfFire {

    sleep 0.5
    p1 tp XYZ(-2133.739990234375, -14999.3896484375, 2.417327880859375)

            break
        } else {
            sleep 1
        }
    }
    while p1 windowvisible ['WorldView', 'windowHUD', 'windowForTopRowButtons', 'containerForTeamUp', 'btnTeamUp'] {
        p1 sendkey X, 0.1
    }
    call load
    print entered dungeon 
}
